B) es geht um die Binäresuche 

#include <stdio.h>

// Funktion für die binäre Suche
int binaereSuche(int arr[], int n, int gesucht) {
    int links = 0;         // Startindex
    int rechts = n - 1;    // Endindex
    
    while (links <= rechts) {
        int mitte = links + (rechts - links) / 2;  // Berechnung des mittleren Index
        
        // Überprüfen, ob der gesuchte Wert gleich dem Mittelwert ist
        if (arr[mitte] == gesucht) {
            return mitte;  // Element gefunden, Index zurückgeben
        }
        
        // Wenn der gesuchte Wert kleiner als der Mittelwert ist, wird die rechte Hälfte ausgeschlossen
        if (arr[mitte] > gesucht) {
            rechts = mitte - 1;
        }
        // Wenn der gesuchte Wert größer als der Mittelwert ist, wird die linke Hälfte ausgeschlossen
        else {
            links = mitte + 1;
        }
    }
    
    return -1;  // Element nicht gefunden, Rückgabe von -1
}
